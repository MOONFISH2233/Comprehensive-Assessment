# 综测加分自评向导 · 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 做一个手机端网页，让学生逐项回答约 129 屏（覆盖细则中全部 159 个具名项目与选项），自动算出五育加分并生成 5 个符合模板格式的 PDF。

**Architecture:** 4 个静态文件（页面 / 项目数据 / 计分引擎 / 应用逻辑）。计分引擎是纯函数、零依赖，用 Node 内置测试跑 TDD；UI 与 PDF 生成靠浏览器实测。图片全程留在浏览器内，服务器只托管空工具页面。

**Tech Stack:** 原生 HTML/CSS/JS（ES modules）、jsPDF（CDN）、Node 内置 `node --test`（仅测试用，无运行时依赖）

**Spec:** `D:\学习\综测\大一下\综测向导\设计文档.md`

## Global Constraints

- **计分取值一律以学院细则为准**（`国家卓越工程师学院2026下学期学生综合测评奖励加分项目实施细则.pdf`），与学校办法冲突处按设计文档第 3 节表格取值，界面上标黄提示。
- **只算加分，不算总分**。`z1`（学业加权平均分）、`t1`（体测成绩）、`m1`/`l1`/`d1`（基础分）一律不参与计算。
- **不做任何网络请求**，不上传任何用户数据。所有 `fetch`/XHR 均为禁止项。
- **文件名用英文**，不可被猜到（服务器无访问控制）。
- **手机优先**：所有交互按 390×780 视口设计，触控目标不小于 44px。

`综测向导/test/catalog.test.mjs`：

```js
import { test } from 'node:test';
import assert from 'node:assert/strict';
import { CATALOG, YU } from '../catalog.js';

test('五个育都有分组', () => {
  for (const y of ['D', 'Z', 'T', 'M', 'L']) {
    const groups = CATALOG.filter(g => g.yu === y);
    assert.ok(groups.length > 0, `${y} 育没有任何分组`);
  }
});

test('分组与项目的 id 全局唯一', () => {
  const ids = new Set();
  for (const g of CATALOG) {
    assert.ok(!ids.has(g.id), `重复的分组 id: ${g.id}`);
    ids.add(g.id);
    for (const it of (g.items ?? [])) {
      assert.ok(!ids.has(it.id), `重复的项目 id: ${it.id}`);
      ids.add(it.id);
    }
  }
});

// 各 count / items-grade 组的具名项目数，逐组钉死。
// 抄录时先按细则原文数出真实数字，填进来，再跑测试。
// 这样以后任何人改动 catalog 漏掉一项，测试会立刻指出是哪一组。
const EXPECTED_ITEMS = {
  'D-sizheng': 7,        // 细则 p1 思政教育加分
  'Z-kejihuodong': 50,   // 细则 p6–7 参加科技学术活动加分
  'T-duanlian': 13,      // 细则 p9 课外体育锻炼活动
  'T-bisai': 11,         // 细则 p9 体育比赛
  'M-huodong': 25,       // 细则 p10–11 美育实践或文艺活动
  'L-sushe': 2,          // 细则 p11 宿舍文明卫生
  'L-shijian': 15        // 细则 p12 社会实践 13 项 + 暑假实习/助教 2 项（0.15 分）
};

test('每一组的项目数与细则原文一致', () => {
  for (const [gid, n] of Object.entries(EXPECTED_ITEMS)) {
    const g = CATALOG.find(x => x.id === gid);
    assert.ok(g, `catalog 里找不到分组 ${gid}`);
    assert.equal(g.items?.length ?? 0, n, `${gid} 项目数应为 ${n}`);
  }
});

test('具名项目总数', () => {
  const n = CATALOG.reduce((s, g) => s + (g.items?.length ?? 0), 0);
  const expected = Object.values(EXPECTED_ITEMS).reduce((a, b) => a + b, 0);
  assert.equal(n, expected, `项目总数应为 ${expected}，实际 ${n}`);
});

test('非 count 组的选项数', () => {
  const asOptions = (g) =>
    g.mode === 'pickmax' ? (g.subgroups ?? []).reduce((s, sg) => s + sg.options.length, 0)
    : (g.options?.length ?? 0);
  assert.equal(asOptions(CATALOG.find(g => g.id === 'D-shehui')), 9);
  assert.equal(asOptions(CATALOG.find(g => g.id === 'D-koufen')), 8);
  assert.equal(asOptions(CATALOG.find(g => g.id === 'Z-jineng')), 15);
});

test('每个分组都有计分模式与封顶字段', () => {
  for (const g of CATALOG) {
    assert.ok(g.mode, `${g.id} 缺少 mode`);
    assert.ok('cap' in g, `${g.id} 缺少 cap 字段（不封顶请显式写 null）`);
  }
});

test('五育元信息完整', () => {
  assert.deepEqual(Object.keys(YU).sort(), ['D', 'L', 'M', 'T', 'Z']);
});
```

```js
import { CATALOG } from './catalog.js';

// 浮点累加会出 0.30000000000000004，统一按「百分位整数」累加，最后再除回来
const toCenti = (n) => Math.round(n * 10000) / 100;
const fromCenti = (c) => Math.round(c) / 100;

function groupScore(group, payload) {
  if (!payload) return { raw: 0, lines: [] };
  const lines = [];

  switch (group.mode) {
    case 'count': {
      for (const item of group.items ?? []) {
        const n = Number(payload[item.id] ?? 0);
        if (n > 0) {
          // 单项可以用 unit 覆盖分组默认分值（如劳育的暑假实习 0.15 分）
          const u = item.unit ?? group.unit;
          lines.push({ itemId: item.id, itemName: item.name ?? item.id, points: u * n });
        }
      }
      break;
    }
    default:
      return { raw: 0, lines: [] };
  }

  let raw = lines.reduce((s, l) => s + l.points, 0);
  let capped = raw;
  if (group.cap != null && raw > group.cap) capped = group.cap;
  return { raw, capped, lines, truncated: capped < raw };
}

export function computeBreakdown(answers, catalog = CATALOG) {
  const out = {};
  for (const group of catalog) {
    const r = groupScore(group, answers[group.id]);
    (out[group.yu] ??= { total: 0, groups: [] }).groups.push({
      groupId: group.id, groupTitle: group.title, mode: group.mode,
      points: r.capped, raw: r.raw, truncated: !!r.truncated,
      cap: group.cap ?? null, lines: r.lines
    });
  }
  return out;
}

export function computeScores(answers, catalog = CATALOG) {
  const bd = computeBreakdown(answers, catalog);
  const out = { D: 0, Z: 0, T: 0, M: 0, L: 0 };
  for (const [yu, v] of Object.entries(bd)) {
    out[yu] = fromCenti(v.groups.reduce((s, g) => s + toCenti(g.points), 0));
  }
  return out;
}
```

```js
export function load() {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    if (raw) Object.assign(state, JSON.parse(raw));
  } catch { /* 损坏的存档直接忽略，从零开始 */ }
}

export function reset() {
  state.answers = {}; state.proofs = {}; state.view = 'overview'; state.cursor = null;
  localStorage.removeItem(STORAGE_KEY);
```

```js
// 把一个分组展开成若干「问题」。
// count 与 items-grade 是「逐项问」（每个具名项目一屏）；
// 其余模式（grade / honor / range / pickmax / tier / penalty）整组算一屏。
export function flattenQuestions(catalog = CATALOG) {
  const out = [];
  for (const group of catalog) {
    if (group.mode === 'count' || group.mode === 'items-grade') {
      for (const item of group.items ?? []) out.push({ group, item, kind: 'item' });
    } else {
      out.push({ group, item: null, kind: 'group' });
    }
  }
  return out;
}
```

```js
function answerOf(group, item) {
  const a = state.answers[group.id];
  if (group.mode === 'count')   return !!(a && a[item.id] > 0);
  if (group.mode === 'items-grade') return !!(a && a[item.id]);
  return !!a;
}

function setAnswer(group, item, on, count = 1) {
  if (group.mode === 'count') {
    const a = state.answers[group.id] ??= {};
    if (on) a[item.id] = count; else delete a[item.id];
  } else if (group.mode === 'items-grade') {
    const a = state.answers[group.id] ??= {};
    if (on) a[item.id] = { levelKey: group.levels[0].key, rankIndex: 0, isExtra: false };
    else delete a[item.id];
  } else {
    if (on) state.answers[group.id] ??= (group.mode === 'grade' ? [] : {});
    else delete state.answers[group.id];

  save();
}
```

```js
export function collectMaterialList() {
  const out = [];
  for (const group of CATALOG) {
    const payload = state.answers[group.id];
    if (!payload) continue;
    if (group.mode === 'count') {
      for (const it of group.items ?? []) {
        const n = Number(payload[it.id] ?? 0);
        if (n > 0) out.push({ yu: group.yu, groupId: group.id, itemId: it.id, itemName: it.name, count: n, points: group.unit * n });
      }
    } else {
      const bd = computeBreakdown({ [group.id]: payload }, [group])[group.yu];
      for (const line of (bd?.groups?.[0]?.lines ?? [])) {
        out.push({ yu: group.yu, groupId: group.id, itemId: line.itemId, itemName: line.itemName, count: 1, points: line.points });
      }
    }
  }
  return out;
}
```

```js
      const scale = Math.min(1, MAX_EDGE / Math.max(img.width, img.height));
      const w = Math.round(img.width * scale);
      const h = Math.round(img.height * scale);
      const cv = document.createElement('canvas');
      cv.width = w; cv.height = h;
      cv.getContext('2d').drawImage(img, 0, 0, w, h);
      resolve({ dataUrl: cv.toDataURL('image/jpeg', JPEG_Q), w, h });
    };
    img.onerror = () => { URL.revokeObjectURL(url); reject(new Error('图片读取失败：' + file.name)); };
    img.src = url;
  });
}
```

```js
function generateYuPdf(yuKey) {
  const jsPDFctor = window.jspdf?.jsPDF;
  if (!jsPDFctor) { alert('PDF 库没加载出来。检查网络后刷新页面重试。'); return; }

  const materials = collectMaterialList().filter(m => m.yu === yuKey);
  const withProofs = materials.filter(m => (state.proofs[m.itemId] ?? []).length > 0);
  if (!withProofs.length) { alert('这一育还没选任何图片，先点「选图」。'); return; }

  const missing = materials.length - withProofs.length;
  if (missing > 0 && !confirm(`这一育还有 ${missing} 项没选图，这些不会出现在 PDF 里。继续生成吗？`)) return;

  const doc = new jsPDFctor({ unit: 'mm', format: 'a4' });
  const PW = 210, PH = 297, M = 15, CW = PW - M * 2;
  let y = M;

  const scores = computeScores(state.answers);
  const yuName = YU[yuKey].name;

  // 标题
  doc.setFontSize(16); doc.setFont(undefined, 'bold');
  doc.text(`${yuName}加分 = ${scores[yuKey].toFixed(2)}`, PW / 2, y, { align: 'center' });
  y += 10;

  let imgNo = 0;
  const groups = new Map();
  for (const m of withProofs) {
    if (!groups.has(m.groupId)) groups.set(m.groupId, []);
    groups.get(m.groupId).push(m);
  }

  let secNo = 0;
  for (const [, list] of groups) {
    secNo++;
    const title = CATALOG.find(g => g.id === list[0].groupId)?.title ?? '';
    if (y > PH - M - 30) { doc.addPage(); y = M; }
    doc.setFontSize(13); doc.setFont(undefined, 'bold');
    doc.text(`${secNo}. ${title}`, PW / 2, y, { align: 'center' });
    y += 8;

    for (const m of list) {
      const imgs = state.proofs[m.itemId];
      for (const im of imgs) {
        imgNo++;
        const h = CW * (im.h / im.w);
        const need = 8 + Math.min(h, PH - M * 2 - 16) + 4;
        if (y + need > PH - M) { doc.addPage(); y = M; }
        doc.setFontSize(11); doc.setFont(undefined, 'normal');
        doc.text(`【图片${imgNo}】${m.itemName}，加 ${m.points.toFixed(2)} 分`, M, y);
        y += 5;
        const maxH = PH - M - y - 4;
        const drawH = Math.min(h, maxH);
        const drawW = drawH * (im.w / im.h);
        doc.addImage(im.dataUrl, 'JPEG', M + (CW - drawW) / 2, y, drawW, drawH);
        y += drawH + 6;
      }
    }
  }

  doc.save(`${yuName}加分-${scores[yuKey].toFixed(2)}.pdf`);
}
```
